ПОЯСНЕНИЕ К ЛИСТИНГУ
------------------------------------------------------------------------------
Исходный текст программы разделён по подсистемам. Каждый файл предваряется
заголовком с порядковым номером, указанием подсистемы и строкой НАЗНАЧЕНИЕ.

Часть текста программы содержит сведения, составляющие коммерческую тайну
правообладателя (Федеральный закон от 29.07.2004 № 98-ФЗ «О коммерческой
тайне»). Эти сведения в настоящем листинге закрыты:
  - файлы, целиком составляющие коммерческую тайну, заменены пометкой с
    указанием объёма закрытого текста; наименования таких файлов не приводятся;
  - отдельные строки открытых файлов заменены пометкой «ЗАКРЫТО» с указанием
    числа закрытых строк.
Оригинальный полный текст программы хранится у правообладателя.

В листинг не включены служебные файлы сборки и сторонние библиотеки.

// =============================================================================
//  chat_store.cpp  —  Персистентный чат в NVS (см. chat_store.h)
// =============================================================================
#include "chat_store.h"
#include <Preferences.h>
#include <Arduino.h>
#include <string.h>

ChatStore chatStore;
static Preferences prefs;

// Компактная запись на одно сообщение
struct StoredMsg {
    char        text[CHAT_STORE_TEXT_MAX + 1];
    mnmp_addr_t addr;
    uint16_t seq;
    uint8_t  outgoing;
    uint8_t  acked;
    int8_t   rssi;
    int8_t   snr;
};

void ChatStore::begin()
{
    prefs.begin("slm_chat", false);
    Serial.printf("[CHAT-NVS] begin, stored=%u\n",
                  (unsigned)prefs.getBytesLength("log") / (unsigned)sizeof(StoredMsg));
}

void ChatStore::load(ChatMsg* log, int& count, int capacity)
{
    size_t len = prefs.getBytesLength("log");
    if (len == 0 || (len % sizeof(StoredMsg)) != 0) { return; }

    int n = (int)(len / sizeof(StoredMsg));
    if (n > capacity)      n = capacity;
    if (n > CHAT_STORE_N)  n = CHAT_STORE_N;

    StoredMsg buf[CHAT_STORE_N];
    prefs.getBytes("log", buf, (size_t)n * sizeof(StoredMsg));

    for (int i = 0; i < n; i++) {
        ChatMsg& m = log[i];
        m = ChatMsg{};
        strncpy(m.text, buf[i].text, CHAT_TEXT_MAX); m.text[CHAT_TEXT_MAX] = '\0';
        m.addr     = buf[i].addr;
        m.seq      = buf[i].seq;
        m.outgoing = buf[i].outgoing != 0;
        m.acked    = buf[i].acked != 0;
        m.rssi     = (float)buf[i].rssi;
        m.snr      = (float)buf[i].snr;
        m.ts       = 0;   // время прошлой сессии не восстановимо
    }
    count = n;
    Serial.printf("[CHAT-NVS] Loaded %d msgs\n", n);
}

void ChatStore::tick(const ChatMsg* log, int count)
{
    if (!_dirty) return;
    unsigned long now = millis();
    if (now - _lastSaveTs < 5000) return;   // throttle: береги flash
    _lastSaveTs = now;
    _dirty = false;
    _save(log, count);
}

void ChatStore::_save(const ChatMsg* log, int count)
{
    int n = count;
    if (n > CHAT_STORE_N) n = CHAT_STORE_N;
    if (n <= 0) return;

    StoredMsg buf[CHAT_STORE_N];
    for (int i = 0; i < n; i++) {
        memset(&buf[i], 0, sizeof(StoredMsg));
        strncpy(buf[i].text, log[i].text, CHAT_STORE_TEXT_MAX);
        buf[i].addr     = log[i].addr;
        buf[i].seq      = log[i].seq;
        buf[i].outgoing = log[i].outgoing ? 1 : 0;
        buf[i].acked    = log[i].acked ? 1 : 0;
        buf[i].rssi     = (int8_t)log[i].rssi;
        buf[i].snr      = (int8_t)log[i].snr;
    }
    uint32_t t0 = micros();
    prefs.putBytes("log", buf, (size_t)n * sizeof(StoredMsg));
    uint32_t dt = micros() - t0;
    if (dt > 50000) Serial.printf("[CHAT-NVS] SLOW putBytes: %lu us\n", (unsigned long)dt);
    Serial.printf("[CHAT-NVS] Saved %d msgs\n", n);
}

void ChatStore::erase()
{
    prefs.remove("log");
    Serial.println("[CHAT-NVS] Erased");
}

ФАЙЛ 2. chat_store.h - подсистема «прикладной уровень»
НАЗНАЧЕНИЕ: интерфейс энергонезависимого хранения журнала сообщений.

#pragma once
// =============================================================================
//  chat_store.h  —  Персистентный чат в NVS (донор B v1.8, адаптировано
//  под ChatMsg/MNMP-адресацию v2 — см. chat_types.h)
//
//  Хранит последние CHAT_STORE_N сообщений между перезагрузками.
//  Запись отложенная (throttle): NVS-flash изнашивается, пишем не чаще
//  одного раза в 5 секунд и только при изменениях.
// =============================================================================

#include <stdint.h>
#include "chat_types.h"

#define CHAT_STORE_N        10
#define CHAT_STORE_TEXT_MAX 200

class ChatStore {
public:
    void begin();                                          // Preferences init
    void load(ChatMsg* log, int& count, int capacity);     // NVS -> RAM
    void scheduleSave() { _dirty = true; }
    void tick(const ChatMsg* log, int count);              // throttled RAM -> NVS
    void erase();                                          // очистка лога в NVS

private:
    bool          _dirty      = false;
    unsigned long _lastSaveTs = 0;
    void _save(const ChatMsg* log, int count);
};

extern ChatStore chatStore;

==============================================================================
ФАЙЛ 3. chat_types.h - подсистема «прикладной уровень»
НАЗНАЧЕНИЕ: типы данных сообщений чата.
==============================================================================
#pragma once
// =============================================================================
// [ЗАКРЫТО: строк - 1; коммерческая тайна]
//
//  Адресация — MNMP (1 байт, см. net/mnmp.h): addr — это src для входящих
//  и dst для исходящих сообщений.
// =============================================================================

#include <stdint.h>
#include "../net/mnmp.h"

#define CHAT_LOG_SIZE  50                  // глубина лога в RAM
#define CHAT_TEXT_MAX  MNMP_MAX_PAYLOAD    // = 200, см. net/mnmp.h

// scope — ключ раскладки по разговорам в UI:
//   0 = общий чат (broadcast, dst=0x00, gid=0)
//   1 = личный    (unicast, dst=конкретный адрес, gid=0)
//   2 = группа    (broadcast, gid!=0; ключ разговора = gid)
// peer всегда в addr (входящие: src; исходящие: dst): для личного разговора
// ключ = addr, для общего — «general», для группы — gid.
#define CHAT_SCOPE_BROADCAST 0
#define CHAT_SCOPE_UNICAST   1
#define CHAT_SCOPE_GROUP     2

struct ChatMsg {
    char     text[CHAT_TEXT_MAX + 1] = {0};
    mnmp_addr_t addr  = 0;     // входящие: src; исходящие: dst (это «собеседник»)
    uint8_t  scope    = CHAT_SCOPE_BROADCAST; // см. CHAT_SCOPE_*
    uint8_t  gid      = 0;     // id группы (0 = не группа)
    uint16_t seq      = 0;
    bool     outgoing = false;
    bool     acked    = false; // зарезервировано: mesh-ARQ (v2.1)
    float    rssi     = 0.0f;
    float    snr      = 0.0f;
    uint32_t ts       = 0;     // millis() на момент добавления
};

ФАЙЛ 4. chat.cpp - подсистема «прикладной уровень»
НАЗНАЧЕНИЕ: реализация чат-модуля: приём, отправка, подтверждения и очередь сообщений.

// =============================================================================
//  chat.cpp  —  см. chat.h
// =============================================================================
#include "chat.h"
#include "chat_store.h"
#include <Arduino.h>
#include <Preferences.h>
#include <string.h>

ChatLog chatLog;

void ChatLog::begin()
{
    chatStore.begin();
    chatStore.load(_log, _count, CHAT_LOG_SIZE);
    _loadOutbox();
    _loadUnread();
}

void ChatLog::tick(uint32_t nowMs)
{
    chatStore.tick(_log, _count);
    // Доставка отложенных личных сообщений: пробуем раз в ~3с (узлы появляются).
    if (_outCount > 0 && (uint32_t)(nowMs - _lastFlushMs) >= 3000U) {
        _lastFlushMs = nowMs;
        _flushOutbox(nowMs);
    }
    if (_unreadDirty) _saveUnread();   // дебаунс: пишем в NVS раз за tick, а не на кадр
}

void ChatLog::_append(const ChatMsg& m)
{
    if (_count < CHAT_LOG_SIZE) {
        _log[_count++] = m;
    } else {
        memmove(&_log[0], &_log[1], sizeof(ChatMsg) * (CHAT_LOG_SIZE - 1));
        _log[CHAT_LOG_SIZE - 1] = m;
    }
    chatStore.scheduleSave();
    if (!m.outgoing) _bumpUnread(m);   // Столп F: входящее → +1 к непрочитанным разговора
}

// ── Столп F: непрочитанные ───────────────────────────────────────────────────
// Ключ разговора: личный → addr собеседника, группа → gid, общий → 0.
static inline mnmp_addr_t convKey(uint8_t scope, mnmp_addr_t addr, uint8_t gid)
{
    if (scope == CHAT_SCOPE_UNICAST) return addr;
    if (scope == CHAT_SCOPE_GROUP)   return (mnmp_addr_t)gid;
    return 0;
}

void ChatLog::_bumpUnread(const ChatMsg& m)
{
    mnmp_addr_t key = convKey(m.scope, m.addr, m.gid);
    for (auto& u : _unread)
        if (u.used && u.scope == m.scope && u.key == key) {
            if (u.count < 0xFFFF) u.count++;
            _unreadDirty = true; return;
        }
    for (auto& u : _unread)
        if (!u.used) { u.used = true; u.scope = m.scope; u.key = key; u.count = 1;
                       _unreadDirty = true; return; }
    // таблица полна (>UNREAD_MAX активных разговоров) — игнорируем счётчик нового
}

uint16_t ChatLog::unreadTotal() const
{
    uint32_t t = 0;
    for (auto& u : _unread) if (u.used) t += u.count;
    return (uint16_t)(t > 0xFFFF ? 0xFFFF : t);
}

uint16_t ChatLog::unreadFor(uint8_t scope, mnmp_addr_t key) const
{
    for (auto& u : _unread)
        if (u.used && u.scope == scope && u.key == key) return u.count;
    return 0;
}

void ChatLog::markRead(uint8_t scope, mnmp_addr_t key)
{
    for (auto& u : _unread)
        if (u.used && u.scope == scope && u.key == key) {
            if (u.count) { u.count = 0; _unreadDirty = true; }
            return;
        }
}

void ChatLog::_saveUnread()
{
    Preferences pr; pr.begin("slm_unrd", false);
    pr.putBytes("u", _unread, sizeof(_unread));
    pr.end();
    _unreadDirty = false;
}

void ChatLog::_loadUnread()
{
    Preferences pr; pr.begin("slm_unrd", true);
    size_t n = pr.getBytesLength("u");
    if (n == sizeof(_unread)) pr.getBytes("u", _unread, sizeof(_unread));
    pr.end();
}

// scope из dst+gid (единая логика для tx/rx).
static uint8_t scopeOf(mnmp_addr_t dst, uint8_t gid)
{
    if (gid) return CHAT_SCOPE_GROUP;
    return (dst == MNMP_ADDR_BROADCAST) ? CHAT_SCOPE_BROADCAST : CHAT_SCOPE_UNICAST;
}

bool ChatLog::send(mnmp_addr_t dst, uint8_t gid, const char* text, uint8_t len, bool urgent)
{
    if (len > CHAT_TEXT_MAX - 1) len = CHAT_TEXT_MAX - 1;   // +1 байт tag в payload

    // Личное сообщение узлу, которого нет online → в outbox (store-and-forward).
    // Локально показываем как исходящее (acked=false = «ожидает доставки»).
    if (dst != MNMP_ADDR_BROADCAST && gid == 0) {
        const NodeEntry* e = nodeTable.find(dst);
        if (!(e && NodeTable::isOnline(*e, millis()))) {
            // Раньше это решение было немо: ЛС уходило в outbox и НИЧЕГО не
            // печаталось — «не доходит» и «сессия не встала» выглядели
            // одинаково снаружи. Теперь видно ИМЕННО здесь ли застряло.
            Serial.printf("[CHAT] dst=0x%04X not in roster/offline (found=%d) -> outbox\n",
                          (unsigned)dst, e ? 1 : 0);
            _enqueueOutbox(dst, gid, text, len);
            ChatMsg m;
            memcpy(m.text, text, len); m.text[len] = '\0';
            m.addr = dst; m.gid = gid; m.scope = CHAT_SCOPE_UNICAST;
            m.outgoing = true; m.acked = false; m.ts = millis();
            _append(m);
            return true;
        }
    }

    uint8_t buf[1 + CHAT_TEXT_MAX];
    buf[0] = gid;                                          // tag: 0=общий/личный, 1..254=группа
    memcpy(buf + 1, text, len);
    // [ЗАКРЫТО: строк - 2; коммерческая тайна]
    // сам поставил байты в _fsPending и дошлёт их, когда сессия встанет —
    // сообщение реально уйдёт в эфир. Раньше этот путь возвращал false БЕЗ
    // _append() — своя история отправителя молчала о нём, хотя получатель
    // его увидит: асимметрия "то отправляется, то нет" (репорт юзера) была
    // именно здесь, а не в потере пакета.
    bool sentNow = mnmpBridge.send(dst, buf, (uint8_t)(len + 1), urgent);

    ChatMsg m;
    memcpy(m.text, text, len); m.text[len] = '\0';
    m.addr     = dst;
    m.gid      = gid;
    m.scope    = scopeOf(dst, gid);
    m.outgoing = true;
    m.acked    = sentNow;                                  // false = ждёт KEX, не потеряно
    m.ts       = millis();
    _append(m);
    return true;
}

// ── Outbox: накопление и доставка отложенных личных сообщений ───────────────
void ChatLog::_enqueueOutbox(mnmp_addr_t dst, uint8_t gid, const char* text, uint8_t len)
{
    if (len > OUTBOX_TEXT_MAX) len = OUTBOX_TEXT_MAX;
    OutboxMsg om;
    om.dst = dst; om.gid = gid; om.ts = millis();
    memcpy(om.text, text, len); om.text[len] = '\0';
    if (_outCount < OUTBOX_MAX) {
        _outbox[_outCount++] = om;
    } else {
        memmove(&_outbox[0], &_outbox[1], sizeof(OutboxMsg) * (OUTBOX_MAX - 1));
        _outbox[OUTBOX_MAX - 1] = om;                      // вытесняем самое старое
    }
    _outboxDirty = true;
    _saveOutbox();
}

void ChatLog::_flushOutbox(uint32_t nowMs)
{
    for (int i = 0; i < _outCount; ) {
        const NodeEntry* e = nodeTable.find(_outbox[i].dst);
        if (e && NodeTable::isOnline(*e, nowMs)) {
            uint8_t l = (uint8_t)strnlen(_outbox[i].text, OUTBOX_TEXT_MAX);
            uint8_t buf[1 + OUTBOX_TEXT_MAX];
            buf[0] = _outbox[i].gid;
            memcpy(buf + 1, _outbox[i].text, l);
            if (mnmpBridge.send(_outbox[i].dst, buf, (uint8_t)(l + 1), false)) {
                mnmp_addr_t dst = _outbox[i].dst;
                memmove(&_outbox[i], &_outbox[i + 1],
                        sizeof(OutboxMsg) * (_outCount - i - 1));
                _outCount--; _outboxDirty = true;
                // Снять «ожидает» с локальных личных сообщений этому адресату.
                for (int j = 0; j < _count; j++)
                    if (_log[j].outgoing && !_log[j].acked &&
                        _log[j].scope == CHAT_SCOPE_UNICAST && _log[j].addr == dst)
                        _log[j].acked = true;
                chatStore.scheduleSave();
                continue;                                  // i не двигаем — сдвинулись
            }
        }
        i++;
    }
    if (_outboxDirty) { _saveOutbox(); _outboxDirty = false; }
}

void ChatLog::_saveOutbox()
{
    Preferences pr; pr.begin("outbox", false);
    pr.putBytes("q", _outbox, sizeof(OutboxMsg) * _outCount);
    pr.putUChar("n", (uint8_t)_outCount);
    pr.end();
}

void ChatLog::_loadOutbox()
{
    Preferences pr; pr.begin("outbox", true);
    int n = pr.getUChar("n", 0);
    if (n > OUTBOX_MAX) n = OUTBOX_MAX;
    size_t len = pr.getBytesLength("q");
    if (n > 0 && len == sizeof(OutboxMsg) * (size_t)n) {
        pr.getBytes("q", _outbox, len);
        _outCount = n;
    }
    pr.end();
}

bool ChatLog::sendGroupInfo(const Group& g)
{
    // [ЗАКРЫТО: строк - 2; коммерческая тайна]
    // >254 усекался бы и «кто в группе» показывалось неверно.
    uint8_t buf[1 + 1 + 1 + NODE_NAME_MAX + 1 + GROUP_MEMBERS_MAX * MNMP_ADDR_SIZE];
    uint8_t p = 0;
    buf[p++] = GROUP_GID_CONTROL;
    buf[p++] = g.gid;
    uint8_t nlen = (uint8_t)strnlen(g.name, NODE_NAME_MAX);
    buf[p++] = nlen;
    memcpy(buf + p, g.name, nlen); p += nlen;
    buf[p++] = g.memberCnt;
    for (uint8_t i = 0; i < g.memberCnt; i++) {
// [ЗАКРЫТО: строк - 1; коммерческая тайна]
        buf[p++] = (uint8_t)(g.members[i] >> 8);
#endif
        buf[p++] = (uint8_t)g.members[i];
    }
    return mnmpBridge.send(MNMP_ADDR_BROADCAST, buf, p, false);
}

void ChatLog::onReceive(const MnmpPacket& pkt, float rssi)
{
    // [ЗАКРЫТО: строк - 1; коммерческая тайна]
    if (pkt.payLen < 1) return;
    uint8_t tag = pkt.payload[0];

    // Control (tag=0xFF): определение группы — не сообщение, а membership-инфо.
    if (tag == GROUP_GID_CONTROL) {
        const uint8_t* d = pkt.payload + 1;
        int n = (int)pkt.payLen - 1;
        if (n < 3) return;                                 // [gid][nameLen][cnt] минимум
        uint8_t gid  = d[0];
        uint8_t nlen = d[1]; if (nlen > NODE_NAME_MAX) nlen = NODE_NAME_MAX;
        if (2 + (int)nlen + 1 > n) return;
        char name[NODE_NAME_MAX + 1] = {0};
        memcpy(name, d + 2, nlen); name[nlen] = '\0';
        uint8_t cnt = d[2 + nlen];
        int avail = n - (2 + (int)nlen + 1);
        int fit   = (avail < 0) ? 0 : avail / MNMP_ADDR_SIZE;   // сколько адресов влезло
        if (cnt > fit) cnt = (uint8_t)fit;
        if (cnt > GROUP_MEMBERS_MAX) cnt = GROUP_MEMBERS_MAX;
        const uint8_t* mp = d + 2 + nlen + 1;
        mnmp_addr_t mem[GROUP_MEMBERS_MAX] = {0};
        for (uint8_t i = 0; i < cnt; i++) {
// [ЗАКРЫТО: строк - 1; коммерческая тайна]
            mem[i] = (mnmp_addr_t)(((uint16_t)mp[i * 2] << 8) | mp[i * 2 + 1]);
#else
            mem[i] = (mnmp_addr_t)mp[i];
#endif
        }
        groupTable.upsert(gid, name, mem, cnt);
        return;
    }

    // Сообщение: tag = gid (0 = не группа). Группу показываем только если я в ней.
    uint8_t gid = tag;
    if (gid != 0 && !groupTable.isMember(gid)) return;

    ChatMsg m;
    uint8_t len = (uint8_t)(pkt.payLen - 1);
    if (len > CHAT_TEXT_MAX) len = CHAT_TEXT_MAX;
    memcpy(m.text, pkt.payload + 1, len); m.text[len] = '\0';
    m.addr  = pkt.src;
    m.gid   = gid;
    m.scope = scopeOf(pkt.dst, gid);
    m.seq   = pkt.seq;
    m.rssi  = rssi;
    m.ts    = millis();
    chatLog._append(m);
}

void ChatLog::erase()
{
    _count = 0;
    chatStore.erase();
}

ФАЙЛ 5. chat.h - подсистема «прикладной уровень»

#pragma once
// =============================================================================
// [ЗАКРЫТО: строк - 1; коммерческая тайна]
//
//  Лог сообщений в RAM (CHAT_LOG_SIZE, см. chat_types.h), персист последних
//  CHAT_STORE_N через app/chat_store. Передача — через net/mnmp_bridge
// [ЗАКРЫТО: строк - 2; коммерческая тайна]
// =============================================================================

#include "chat_types.h"
#include "../net/mnmp_bridge.h"
#include "../net/node_table.h"
#include "group_table.h"

// теряются — копятся здесь (NVS-персист) и доставляются, как только узел

static Preferences gprefs;

void GroupTable::begin(mnmp_addr_t selfAddr)
{
    _self = selfAddr;
    _load();
}

Group* GroupTable::_find(uint8_t gid)
{
    for (int i = 0; i < GROUP_MAX; i++)
        if (_g[i].gid == gid) return &_g[i];
    return nullptr;
}

const Group* GroupTable::find(uint8_t gid) const
{
    for (int i = 0; i < GROUP_MAX; i++)
        if (_g[i].gid == gid) return &_g[i];
    return nullptr;
}

Group* GroupTable::_freeSlot()
{
    for (int i = 0; i < GROUP_MAX; i++)
        if (_g[i].gid == 0) return &_g[i];
    return nullptr;
}

// Добавить себя в участники (если ещё нет и есть место) — создатель всегда член.
void GroupTable::_addSelf(Group* g)
{
    for (uint8_t i = 0; i < g->memberCnt; i++)
        if (g->members[i] == _self) return;
    if (g->memberCnt < GROUP_MEMBERS_MAX) g->members[g->memberCnt++] = _self;
}

uint8_t GroupTable::create(const char* name, const mnmp_addr_t* members, uint8_t cnt)
{
    Group* g = _freeSlot();
    if (!g) return 0;                                   // нет места

    // Свободный gid 1..254, не занятый локально и НЕ из зарезервированных тегов
    // app-слоя (config.h): 0xFD стресс-тест, 0xFE MAVLink-мост, 0xFF control. Группа
    // с таким gid уехала бы в чужой диспетчер appRx (её сообщения — не в чат).
    uint8_t gid = 0;
    for (int tries = 0; tries < 64; tries++) {
        uint8_t cand = (uint8_t)(1 + (esp_random() % 254)); // 1..254
        if (cand == GROUP_GID_CONTROL || cand == STRESS_TAG || cand == MAV_TAG) continue;
        if (!_find(cand)) { gid = cand; break; }
    }
    if (!gid) return 0;

    *g = Group{};
    g->gid = gid;
    strncpy(g->name, name ? name : "", NODE_NAME_MAX); g->name[NODE_NAME_MAX] = '\0';
    if (cnt > GROUP_MEMBERS_MAX) cnt = GROUP_MEMBERS_MAX;
    for (uint8_t i = 0; i < cnt; i++) g->members[i] = members[i];
    g->memberCnt = cnt;
    _addSelf(g);
    _save();
    return gid;
}

bool GroupTable::upsert(uint8_t gid, const char* name, const mnmp_addr_t* members, uint8_t cnt)
{
    // Отвергаем 0 и зарезервированные теги (config.h) — в т.ч. от соседа со старой
    // прошивкой, где 0xFD/0xFE ещё не были исключены при создании группы.
    if (gid == 0 || gid == GROUP_GID_CONTROL || gid == STRESS_TAG || gid == MAV_TAG)
        return false;
    if (cnt > GROUP_MEMBERS_MAX) cnt = GROUP_MEMBERS_MAX;

    // Принимаем только группы, где я участник.
    bool meIn = false;
    for (uint8_t i = 0; i < cnt; i++) if (members[i] == _self) { meIn = true; break; }
    if (!meIn) return false;

    Group* g = _find(gid);
    if (!g) g = _freeSlot();
    if (!g) return false;                               // нет места

    g->gid = gid;
    strncpy(g->name, name ? name : "", NODE_NAME_MAX); g->name[NODE_NAME_MAX] = '\0';
    for (uint8_t i = 0; i < cnt; i++) g->members[i] = members[i];
    g->memberCnt = cnt;
    _save();
    return true;
}

bool GroupTable::isMember(uint8_t gid) const
{
    const Group* g = find(gid);
    if (!g) return false;
    for (uint8_t i = 0; i < g->memberCnt; i++)
        if (g->members[i] == _self) return true;
    return false;
}

bool GroupTable::leave(uint8_t gid)
{
    Group* g = _find(gid);
    if (!g) return false;
    *g = Group{};
    _save();
    return true;
}

int GroupTable::count() const
{
    int n = 0;
    for (int i = 0; i < GROUP_MAX; i++) if (_g[i].gid != 0) n++;
    return n;
}

void GroupTable::_save()
{
    // [ЗАКРЫТО: строк - 2; коммерческая тайна]
    // старый блоб просто игнорируется, группы создаются заново.
    gprefs.begin("groups", false);
    gprefs.putBytes(GROUP_NVS_KEY, _g, sizeof(_g));
    gprefs.end();
}

void GroupTable::_load()
{
    gprefs.begin("groups", true);
    size_t len = gprefs.getBytesLength(GROUP_NVS_KEY);
    if (len == sizeof(_g)) gprefs.getBytes(GROUP_NVS_KEY, _g, sizeof(_g));
    gprefs.end();
}

#pragma once
// =============================================================================
//  pass_policy.h — правила смены паролей (чистая логика, host-тестируемо).
//
//  ⚠ КОНТЕКСТ, который определяет строгость: в изделии НЕТ сброса к заводским, НЕТ
//  мастер-пароля и НЕТ канала восстановления (решение 2026-07-31). Забыл пароль —
//  прибор списывается. Поэтому политика бьёт не столько по «слабым» паролям,
//  сколько по СПОСОБАМ ПОТЕРЯТЬ ПРИБОР:
//    • обязательное повторное подтверждение (опечатка = безвозвратно);
//    • только печатные ASCII — пароль с кириллицей/невидимым символом наберётся в
//      другой раскладке иначе, а проверить будет уже негде;
//    • длина ≥ 8 для WiFi — это НЕ вкусовщина, WPA2-PSK короче 8 символов
//      невозможен, точка доступа просто не поднимется;
//    • новый пароль не равен заводскому (иначе «смена» ничего не меняет).
// =============================================================================
#include <stdint.h>
#include <string.h>

#define PASS_WIFI_MIN   8    // жёсткий предел WPA2-PSK
#define PASS_WIFI_MAX  63    // жёсткий предел WPA2-PSK
#define PASS_WEB_MIN    6
#define PASS_WEB_MAX   63
// [ЗАКРЫТО: строк - 2; коммерческая тайна]
// произвольной длины) — длиннее 16 молча обрежется, поэтому максимум = 16, а не 63.
#define PASS_PSK_MIN    8
#define PASS_PSK_MAX   16

enum class PassVerdict : uint8_t {
    OK = 0,
    TOO_SHORT,      // короче минимума (для WiFi — WPA2 не примет)
    TOO_LONG,
    NOT_ASCII,      // непечатный/не-ASCII символ
    SAME_AS_OLD,    // равен заводскому/текущему — смены не произошло
    MISMATCH,       // повтор не совпал (защита от опечатки: отката нет)
};

// Проверить новый пароль. confirm — повтор (nullptr = не проверять), old — текущий
// или заводской (nullptr = не сравнивать).
inline PassVerdict passCheck(const char* p, const char* confirm,
                             int minLen, int maxLen, const char* old)
{
    if (!p) return PassVerdict::TOO_SHORT;
    int n = (int)strlen(p);
    if (n < minLen) return PassVerdict::TOO_SHORT;
    if (n > maxLen) return PassVerdict::TOO_LONG;
    for (int i = 0; i < n; i++) {
        unsigned char c = (unsigned char)p[i];
        if (c < 0x20 || c > 0x7E) return PassVerdict::NOT_ASCII;
    }
    if (old && strcmp(p, old) == 0) return PassVerdict::SAME_AS_OLD;
    if (confirm && strcmp(p, confirm) != 0) return PassVerdict::MISMATCH;
    return PassVerdict::OK;
}

// Короткий машинный код причины для JSON-ответа UI.
inline const char* passVerdictCode(PassVerdict v)
{
    switch (v) {
        case PassVerdict::OK:          return "ok";
        case PassVerdict::TOO_SHORT:   return "short";
        case PassVerdict::TOO_LONG:    return "long";
        case PassVerdict::NOT_ASCII:   return "ascii";
        case PassVerdict::SAME_AS_OLD: return "same";
        case PassVerdict::MISMATCH:    return "mismatch";
    }
    return "err";
}

        Serial.println("[SEC] пароль WiFi в NVS негоден → заводской (AP не будет открытой)");
    }
    WiFi.softAP(_ssid, _wifiPass, WIFI_AP_CHANNEL, 0, WIFI_AP_MAX_CONN);
    // §3.6: power-save выключен обязательно — иначе AP «засыпает» между
    // [ЗАКРЫТО: строк - 1; коммерческая тайна]
    esp_wifi_set_ps(WIFI_PS_NONE);
    IPAddress ip = WiFi.softAPIP();
    Serial.printf("[WIFI] AP: %s  IP: %s\n", _ssid, ip.toString().c_str());

    if (MDNS.begin(MDNS_HOST)) Serial.printf("[WIFI] mDNS: http://%s.local\n", MDNS_HOST);
    else                        Serial.println("[WIFI] mDNS init failed (non-fatal)");

    _registerRoutes();
    _srv.begin();
    Serial.printf("[WIFI] HTTP on port %d\n", WEB_SERVER_PORT);
#if RELEASE_HARDENING
    Serial.printf("[SEC] релизный режим: доступ только после входа (%s), провижининг %s\n",
                  WEB_ADMIN_USER, _provDone ? "пройден" : "ТРЕБУЕТСЯ (пароли заводские)");
    Serial.println("[SEC] ⚠ восстановления нет: забытый пароль = прибор списывается");
#endif
}

void AppWebServer::loop()
{
    uint32_t t0 = micros();
    _srv.handleClient();
    uint32_t dt = micros() - t0;
    if (dt > 50000) Serial.printf("[WEB] SLOW handleClient: %lu us\n", (unsigned long)dt);
}

// Гвард доступа. Порядок важен: СНАЧАЛА авторизация (до неё не отдаём ничего — ни
// чтения, ни записи, решение юзера 2026-07-31), и только потом проверка провижининга.
// Digest: пароль не уходит по сети в открытом виде, диалог рисует сам браузер.
bool AppWebServer::_guard(bool needProvisioned)
{
#if RELEASE_HARDENING
    if (!_srv.authenticate(WEB_ADMIN_USER, _webPass)) {
        _srv.requestAuthentication(DIGEST_AUTH, WEB_AUTH_REALM, "Доступ запрещён");
        return false;
    }
    if (needProvisioned && !_provDone) {
        _srv.send(403, "application/json",
                  "{\"ok\":false,\"err\":\"provision\"}");   // сперва смените пароли
        return false;
    }
#else
    (void)needProvisioned;
#endif
    return true;
}

    snprintf(buf, sizeof(buf), "{\"prov\":%s,\"hard\":%d}",
             _provDone ? "true" : "false", (int)RELEASE_HARDENING);
    _srv.send(200, "application/json", buf);
}

// POST /api/provision {wifi,wifi2,web,web2} — обязательная смена ОБОИХ заводских
// паролей. Повтор каждого обязателен: отката нет, опечатка = потерянный прибор.
void AppWebServer::_handleProvision()
{
    if (!_guard(false)) return;
    JsonDocument doc;
    if (deserializeJson(doc, _srv.arg("plain")) != DeserializationError::Ok) {
        _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"json\"}");
        return;
    }
    const char* w1 = doc["wifi"]  | "";
    const char* w2 = doc["wifi2"] | "";
    const char* b1 = doc["web"]   | "";
    const char* b2 = doc["web2"]  | "";

    PassVerdict vw = passCheck(w1, w2, PASS_WIFI_MIN, PASS_WIFI_MAX, WIFI_AP_PASSWORD);
    if (vw != PassVerdict::OK) {
        char e[80]; snprintf(e, sizeof(e), "{\"ok\":false,\"field\":\"wifi\",\"err\":\"%s\"}",
                             passVerdictCode(vw));
        _srv.send(400, "application/json", e); return;
    }
    PassVerdict vb = passCheck(b1, b2, PASS_WEB_MIN, PASS_WEB_MAX, WEB_ADMIN_PASS);
    if (vb != PassVerdict::OK) {
        char e[80]; snprintf(e, sizeof(e), "{\"ok\":false,\"field\":\"web\",\"err\":\"%s\"}",
                             passVerdictCode(vb));
        _srv.send(400, "application/json", e); return;
    }

    strncpy(_wifiPass, w1, sizeof(_wifiPass) - 1); _wifiPass[sizeof(_wifiPass) - 1] = 0;
    strncpy(_webPass,  b1, sizeof(_webPass)  - 1); _webPass[sizeof(_webPass)  - 1] = 0;
    _provDone = 1;
    {
        Preferences pr; pr.begin("mesh", false);
        pr.putString("wifipass", _wifiPass);
        pr.putString("webpass",  _webPass);
        pr.putUChar("provdone", 1);
        pr.end();
    }
    // Печать ОДИН РАЗ в момент установки: консоль при пуско-наладке есть, а позже
    // взять пароль негде. Это НЕ канал восстановления — прочитать его потом нельзя.
    Serial.printf("[SEC] пароль WiFi установлен: %s  (ЗАПИШИТЕ — восстановления нет)\n",
                  _wifiPass);
    Serial.println("[SEC] пароль веб-интерфейса изменён; перезагрузка");
    _srv.send(200, "application/json", "{\"ok\":true,\"restart\":true}");
    delay(200);
    ESP.restart();          // AP поднимается с новым паролем, клиент переподключается
}

// Имя пресета по текущим sf/bw (read-only показ радио). "—" если не совпало.
static const char* presetName(uint8_t sf, float bw)
{
    for (int i = 0; i < NUM_PRESETS; i++)
        if (LORA_PRESETS[i].sf == sf && LORA_PRESETS[i].bw == bw)
            return LORA_PRESETS[i].name;
    return "—";
}

void AppWebServer::_handleStats()
{
    if (!_guard()) return;
    JsonDocument doc;

    // [ЗАКРЫТО: строк - 2; коммерческая тайна]
    doc["freq"]      = radioMgr.currentFreqMHz();   // физическая частота (не прыгает в JOINING)
    // [ЗАКРЫТО: строк - 1; коммерческая тайна]
    doc["numNodes"]  = discovery.numNodes();
    doc["synced"]    = linkMac.synced();             // RUN = true, JOINING/SYNCWAIT = false
    doc["meshAddr"]  = mnmpBridge.myAddr();
    doc["neighbors"] = mnmpBridge.neighborCount();
    doc["unread"]    = chatLog.unreadTotal();                          // сумма непрочитанных (бейдж «Чаты»)
    doc["unreadGen"] = chatLog.unreadFor(CHAT_SCOPE_BROADCAST, 0);     // непрочит. в общем чате
    // Задача 14 (SIM-SYNTHESIS-REPORT.md §5.1): "видимость перегрузки, не
    // только факт доставки" — та же занятость TX-очереди, что уже считается
    // для admission control (MESH_SEND_ADMISSION_MAX — общая константа,
    // #define всегда виден независимо от того, включён ли сам гейт),
    // просто прокинута в UI как явный сигнал.
    doc["txq"]    = linkMac.txQueued();
    doc["txqMax"] = LinkMac::TXQ_LEN;
    // Стресс-тест (меню «Разработчик»): живой статус мастера.
    JsonObject stObj = doc["stress"].to<JsonObject>();
    stObj["run"]  = stress.running();
    stObj["size"] = stress.curSize();
    stObj["sent"] = stress.sent();
    stObj["ok"]   = stress.ok();
    stObj["lost"] = stress.lost();
#if UAV_MAVLINK
    // Столп C: состояние MAVLink-моста (снимок из main) — для карточки «Разработчик».
    JsonObject mv = doc["mav"].to<JsonObject>();
    mv["peer"]   = _mav.peer;
    mv["tx"]     = _mav.tx;          // кадров MAVLink отправлено в mesh
    mv["rx"]     = _mav.rx;          // собрано из mesh и отдано в UART
    mv["queued"] = _mav.queued;
    mv["arq"]    = _mav.arq;         // команд ждут ACK
    mv["failed"] = _mav.failed;      // команд провалено (лимит ретрансмитов)
    mv["drop"]   = _mav.drop;        // чанков не отдано в mesh (нет маршрута / очередь MAC)
    mv["crcBad"] = _mav.crcBad;      // кадров с UART отброшено по CRC (шлейф/помеха/бод)
    mv["crcUnk"] = _mav.crcUnk;      // msgid вне таблицы CRC_EXTRA (пропущены без проверки)
    mv["thin"]   = _mav.thin;        // телеметрии прорежено политикой (интервал по msgid)
    mv["merge"]  = _mav.merge;       // устаревших кадров заменено свежими в очереди
#endif
#if MESH_MULTINODE
    // [ЗАКРЫТО: строк - 1; коммерческая тайна]
#if MESH_TIME_TREE
    doc["parent"] = (discovery.myRank() == 0) ? 0 : meshAddrFromId(discovery.parentId());
#endif
    JsonArray ign = doc["ignore"].to<JsonArray>();
    for (uint8_t i = 0; i < Discovery::IGNORE_MAX; i++)
        if (discovery.ignored()[i]) ign.add(discovery.ignored()[i]);
#endif
    doc["ssid"]      = _ssid;
    doc["name"]      = _pname;                        // позывное (профиль)

    // Радио (read-only; источник — снимок из main, частота — из radioMgr)
    JsonObject radio = doc["radio"].to<JsonObject>();
    radio["preset"]    = presetName(_sf, _bw);
    radio["idx"]       = radioSwitchState.curPreset;     // активный пресет (для подсветки)
    // Список пресетов из прошивки (UI рендерит по нему; при DUAL_MODEM тут и GFSK).
    JsonArray rpr = radio["presets"].to<JsonArray>();
    for (uint8_t i = 0; i < NUM_PRESETS; i++) {
        JsonObject po = rpr.add<JsonObject>();
        po["name"] = LORA_PRESETS[i].name;
        po["gfsk"] = (LORA_PRESETS[i].mod == MOD_GFSK);   // модуляция: GFSK-профиль
    }
    radio["auto"]      = radioSwitchState.autoMode;
    radio["switching"] = radioSwitchState.pending;       // идёт запланированный свитч
    radio["sf"]        = _sf;
    radio["bw"]        = _bw;
    radio["cr"]        = _cr;
    radio["power"]     = _power;
    radio["powerMin"]  = LORA_POWER_MIN;   // диапазон ползунка мощности (зависит от платы)
    radio["powerMax"]  = LORA_POWER_MAX;

    JsonArray msgs = doc["msgs"].to<JsonArray>();
    int n = chatLog.count();
    constexpr int MAX_SHOWN = 30;
    int from = (n > MAX_SHOWN) ? (n - MAX_SHOWN) : 0;
    for (int i = n - 1; i >= from; i--) {   // новые -> старые
        const ChatMsg& m = chatLog.at(i);
        JsonObject o = msgs.add<JsonObject>();
        o["text"]  = m.text;
        o["addr"]  = m.addr;
        o["scope"] = m.scope;            // 0=общий 1=личный 2=группа
        o["gid"]   = m.gid;              // id группы (0 = не группа)
        o["seq"]   = m.seq;
        o["out"]   = m.outgoing;
        o["pend"]  = m.outgoing && !m.acked;   // в outbox, ждёт доставки (адресат offline)
        o["rssi"]  = (int)m.rssi;
    }

    String out;
    serializeJson(doc, out);
    _srv.send(200, "application/json", out);
}

static const char* roleStr(NodeRole r)

    // [ЗАКРЫТО: строк - 1; коммерческая тайна]
                 case NodeRole::ROUTER: return "router";
                 default:               return "client"; }

// Позиции для UI-карты (Столп E, вариант A): свой РЧ-фикс + известные якоря

// Ростер сети (membership). Соседи = hops==1, «дальше» = hops>1. Включает сам
// узел (self=true) — чтобы UI показывал «все активные модули», включая себя.
void AppWebServer::_handleNodes()
{
    if (!_guard()) return;
    JsonDocument doc;
    uint32_t now = millis();

    JsonArray arr = doc["nodes"].to<JsonArray>();

    JsonObject me = arr.add<JsonObject>();
    me["addr"]  = mnmpBridge.myAddr();
    // [ЗАКРЫТО: строк - 1; коммерческая тайна]
    me["hw"]    = hwStr((NodeHw)_hwType);
    me["name"]  = _pname;
    me["hops"]  = 0;
    me["self"]  = true;
    me["online"]= true;

    const NodeEntry* e = nodeTable.entries();
    // [ЗАКРЫТО: строк - 1; коммерческая тайна]
    // как в serial [MAC] RX ...) приоритетнее устаревшего из ростера (обновляется
    // только на NODE_ANNOUNCE, раз в 10-30с backoff) — для текущих 1-хоповых
    // соседей отдаём именно его, иначе веб показывает RSSI многосекундной давности.
    const NeighborEntry* nbrs = meshRouter.neighbors();
    for (int i = 0; i < MESH_NODES_MAX; i++) {
        if (e[i].id == 0) continue;
        JsonObject o = arr.add<JsonObject>();
        o["addr"]   = e[i].id;
        o["role"]   = roleStr(e[i].role);
        o["hw"]     = hwStr(e[i].hw);
        o["name"]   = e[i].name;
        o["hops"]   = e[i].hops;
        o["next"]   = e[i].nextHop;
        int8_t rssi = e[i].rssi;
        for (int j = 0; j < MESH_NEIGHBOR_MAX; j++)
            if (nbrs[j].addr == e[i].id) { rssi = nbrs[j].rssi; break; }
        o["rssi"]   = rssi;
        o["ageMs"]  = (uint32_t)(now - e[i].lastSeen);
        o["online"] = NodeTable::isOnline(e[i], now);   // offline-контакт всё равно выбираем (outbox)
        o["self"]   = false;
        o["unread"] = chatLog.unreadFor(CHAT_SCOPE_UNICAST, e[i].id);  // бейдж per-контакт
    }
    doc["count"]   = nodeTable.count() + 1;       // всего контактов (+сам)
    doc["online"]  = nodeTable.onlineCount(now) + 1;

    String out;
    serializeJson(doc, out);
    _srv.send(200, "application/json", out);
}

// GET /api/gnss — состояние ГНСС для UI. enabled=false, если прошивка собрана
// без поддержки (GNSS_ENABLE=0). Координаты/фикс показываются всегда; disc —
// рантайм-тумблер дисциплины времени.
void AppWebServer::_handleGnss()
{
    if (!_guard()) return;
    JsonDocument doc;
#if GNSS_ENABLE
    doc["enabled"] = true;
    doc["disc"]    = _gnssDisc;
    doc["fix"]     = gnss.fixValid()      ? 1 : 0;
    doc["lock"]    = timeCore.ppsLocked() ? 1 : 0;
    doc["lat"]     = gnss.lat();
    doc["lon"]     = gnss.lon();
    doc["alt"]     = gnss.altM();
    doc["spd"]     = gnss.speedMps();
    doc["rej"]     = (uint32_t)diag.gnssReject;
    doc["corr"]    = (int32_t)diag.gnssCorr;
    // Диагностика связи (для теста в помещении): raw=байты с UART (0 → молчит,
    // пины/питание/baud); nmea=валидные строки (>0 → модуль ОК); sats; возраст.
    doc["raw"]     = gnss.rawBytes();
    doc["nmea"]    = gnss.nmeaCount();
    doc["sats"]    = gnss.sats();
    doc["rxPin"]   = gnss.rxPin();        // фактический RX-пин (после автоскана)
    doc["nmeaAgeMs"] = gnss.lastNmeaMs() ? (uint32_t)(millis() - gnss.lastNmeaMs()) : 0;
#if GNSS_ANTISPOOF
    // Задача 4: вердикт анти-спуфа (SpoofVerdict) для UI — раньше нигде не
    // отдавался наружу, хотя main.cpp уже ГЕЙТИТ им дисциплину времени и (см.
    // sendNodePos) вещание позиции. 0=OK, 1=NO_PREV(первый фикс), 2..5=отвергнут.
    doc["spoof"]    = (uint8_t)diag.gnssSpoof;
    doc["outvote"]  = (uint8_t)diag.gnssOutvote;
#endif
#else
    doc["enabled"] = false;
#endif
    String out;
    serializeJson(doc, out);
    _srv.send(200, "application/json", out);
}

// POST {hw:0..3} — тип устройства (membership); {gnssDisc:0|1} — ГНСС-дисциплина.
// Персист в NVS; main подхватывает hwType()/gnssDiscipline() в loop.
void AppWebServer::_handleConfig()
{
    if (!_guard()) return;
    JsonDocument doc;
    if (deserializeJson(doc, _srv.arg("plain")) != DeserializationError::Ok) {
        _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"json\"}");
        return;
    }
    Preferences prefs;
    prefs.begin("mesh", false);
    if (doc["hw"].is<int>()) {
        int hw = doc["hw"].as<int>();
        if (hw < 0 || hw > 3) {
            prefs.end();
            _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"hw\"}");
            return;
        }
        _hwType = (uint8_t)hw;
        prefs.putUChar("hwtype", _hwType);
    }
    if (doc["gnssDisc"].is<int>()) {
        _gnssDisc = doc["gnssDisc"].as<int>() ? 1 : 0;
        prefs.putUChar("gnssdisc", _gnssDisc);
    }
#if UAV_MAVLINK
    // {mavPeer:адрес} — куда этот узел отправляет MAVLink с своего UART. 0 = не задан
    // (только приём). main подхватит в мост со следующего loop.
    if (doc["mavPeer"].is<int>()) {
        int mp = doc["mavPeer"].as<int>();
        if (mp < 0 || mp > (int)(mnmp_addr_t)~(mnmp_addr_t)0) {
            prefs.end();
            _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"mavPeer\"}");
            return;
        }
        _mavPeer = (mnmp_addr_t)mp;
// [ЗАКРЫТО: строк - 1; коммерческая тайна]
        prefs.putUShort("mavpeer", (uint16_t)_mavPeer);
#else
        prefs.putUChar("mavpeer", (uint8_t)_mavPeer);
#endif
    }
#endif
    if (doc["name"].is<const char*>()) {
        const char* nm = doc["name"].as<const char*>();
        strncpy(_pname, nm ? nm : "", sizeof(_pname) - 1);
        _pname[sizeof(_pname) - 1] = '\0';
        prefs.putString("pname", _pname);
    }
    // {preset:idx} — инициировать координированный свитч пресета по всей сети.
    if (doc["preset"].is<int>()) {
        int idx = doc["preset"].as<int>();
        if (idx >= 0 && idx < NUM_PRESETS) {
#if RADIO_SWITCH_2PHASE
            // Задача 10: ручной запрос из UI — тот же PREPARE/ACK-кворум, что и
            // авто-режим (main.cpp резолвит commit/abort по итогу окна); не
            // трогаем, если уже идёт другой раунд.
            if (!radioSwitchQuorum.active()) {
                uint8_t prepId = radioSwitchNextPrepId();
                uint32_t nowMs = millis();
                radioSwitchQuorum.begin((uint8_t)idx, prepId, (uint8_t)nodeTable.onlineCount(nowMs), nowMs);
                meshRouter.sendRadioSwitchPrepare((uint8_t)idx, prepId);
            }
#else
            // [ЗАКРЫТО: строк - 3; коммерческая тайна]
#endif
            prefs.putUChar("preset", (uint8_t)idx);                    // sticky
        }
    }
    // [ЗАКРЫТО: строк - 1; коммерческая тайна]
    if (doc["autoPreset"].is<int>()) {
        radioSwitchState.autoMode = doc["autoPreset"].as<int>() != 0;
        prefs.putUChar("automode", radioSwitchState.autoMode ? 1 : 0);
    }
    // {power:дБм} — мощность TX. ЛОКАЛЬНО, отдельно от пресета: применяем сразу к
    // [ЗАКРЫТО: строк - 1; коммерческая тайна]
    // переносит её в cfg на старте и при свитче пресета (txPower()).
    if (doc["power"].is<int>()) {
        int p = doc["power"].as<int>();
        if (p < LORA_POWER_MIN) p = LORA_POWER_MIN;
        if (p > LORA_POWER_MAX) p = LORA_POWER_MAX;
        _txPower = (int8_t)p;
        _power   = _txPower;                 // обновляем read-only показ
        prefs.putChar("txpower", _txPower);
        radioMgr.setPower(_txPower);
    }
#if MESH_MULTINODE
    // Лаб: ручная топология дерева — {ignoreAdd|ignoreDel:addr}|{ignoreClear:1}.
    // «Не слышать» соседа → узел переподключается через другого (тест многохоп Ф1).
    if (doc["ignoreAdd"].is<int>())   discovery.setIgnore((uint8_t)doc["ignoreAdd"].as<int>(), true);
    if (doc["ignoreDel"].is<int>())   discovery.setIgnore((uint8_t)doc["ignoreDel"].as<int>(), false);
    if (doc["ignoreClear"].is<int>()) discovery.clearIgnore();
    if (doc["ignoreAdd"].is<int>() || doc["ignoreDel"].is<int>() || doc["ignoreClear"].is<int>())
        prefs.putBytes("ignore", discovery.ignored(), Discovery::IGNORE_MAX);
#endif
    prefs.end();
    _srv.send(200, "application/json", "{\"ok\":true}");
}

void AppWebServer::_handleSend()
{
    if (!_guard()) return;
    JsonDocument doc;
    if (deserializeJson(doc, _srv.arg("plain")) != DeserializationError::Ok) {
        _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"json\"}");
        return;
    }
    const char* text = doc["text"] | "";
    uint8_t len = (uint8_t)strnlen(text, CHAT_TEXT_MAX);
    if (len == 0) {
        _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"empty\"}");
        return;
    }
    // dst опционален: отсутствует/0 = общий/групповой (broadcast), иначе личный.
    // gid опционален: 0 = не группа, 1..254 = групповой чат (dst тогда broadcast).
    mnmp_addr_t dst = (mnmp_addr_t)(doc["dst"] | 0);
    uint8_t gid = (uint8_t)(doc["gid"] | 0);
    if (gid) dst = MNMP_ADDR_BROADCAST;                  // группа всегда broadcast+фильтр
    bool urgent = doc["urgent"] | false;                 // задача 14: приоритет по требованию юзера
    bool ok = chatLog.send(dst, gid, text, len, urgent);
    _srv.send(200, "application/json", ok ? "{\"ok\":true}" : "{\"ok\":false,\"err\":\"busy\"}");
}

}

// POST /api/stress — {op:start|stop, peer?, n?, period?, sizes?[]} — стресс-тест сети.
// peer не задан → авто (первый узел ростера, в 2-узловой сети это пир). Дефолт-свип
// размеров, если sizes пуст. Меню «Разработчик».
void AppWebServer::_handleStress()
{
    if (!_guard()) return;
    JsonDocument doc;
    if (deserializeJson(doc, _srv.arg("plain")) != DeserializationError::Ok) {
        _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"json\"}");
        return;
    }
    const char* op = doc["op"] | "start";
    if (strcmp(op, "stop") == 0) {
        stress.stop();
        _srv.send(200, "application/json", "{\"ok\":true}");
        return;
    }
    mnmp_addr_t peer = (mnmp_addr_t)(doc["peer"] | 0);
    if (!peer) {                                    // авто-пир: первый контакт из ростера
        const NodeEntry* e = nodeTable.entries();
        for (int i = 0; i < MESH_NODES_MAX; i++) if (e[i].id) { peer = e[i].id; break; }
    }
    if (!peer) { _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"no_peer\"}"); return; }

    uint16_t n      = (uint16_t)(doc["n"]      | 200);
    // [ЗАКРЫТО: строк - 2; коммерческая тайна]
    // периоде тест СОЗНАТЕЛЬНО перегружает пайп (замер потолка), и высокая «потеря»
    // — это переподписка, а не развал сети. Хочешь честный ~0% — period >= 1500.
    uint16_t period = (uint16_t)(doc["period"] | 1500);
    uint16_t sizes[STRESS_MAX_SIZES]; uint8_t ns = 0;
    for (JsonVariant v : doc["sizes"].as<JsonArray>()) {
        if (ns >= STRESS_MAX_SIZES) break;
        sizes[ns++] = (uint16_t)v.as<int>();
    }
    if (!ns) { const uint16_t def[] = {10,50,100,150,200}; for (uint8_t i=0;i<5;i++) sizes[ns++]=def[i]; }
    stress.start(peer, n, period, sizes, ns, millis());
    char resp[48]; snprintf(resp, sizeof(resp), "{\"ok\":true,\"peer\":%u}", (unsigned)peer);
    _srv.send(200, "application/json", resp);
}

// POST /api/chat/read — {gid} | {peer:addr} | {general:true} → пометить разговор
// прочитанным (обнулить счётчик непрочитанных НА УСТРОЙСТВЕ). Столп F.
void AppWebServer::_handleChatRead()
{
    if (!_guard()) return;
    JsonDocument doc;
    if (deserializeJson(doc, _srv.arg("plain")) != DeserializationError::Ok) {

    }
    uint8_t     gid  = (uint8_t)(doc["gid"]  | 0);

    else if (peer) chatLog.markRead(CHAT_SCOPE_UNICAST,   peer);

    _srv.send(200, "application/json", "{\"ok\":true}");
}

// GET /api/groups — список групп, в которых состоит узел.
void AppWebServer::_handleGroupsGet()
{
    if (!_guard()) return;
    JsonDocument doc;
    JsonArray arr = doc["groups"].to<JsonArray>();
    const Group* g = groupTable.groups();
    for (int i = 0; i < GROUP_MAX; i++) {
        if (g[i].gid == 0) continue;
        JsonObject o = arr.add<JsonObject>();
        o["gid"]  = g[i].gid;
        o["name"] = g[i].name;
        o["unread"] = chatLog.unreadFor(CHAT_SCOPE_GROUP, g[i].gid);   // бейдж per-группа
        JsonArray mem = o["members"].to<JsonArray>();
        for (uint8_t k = 0; k < g[i].memberCnt; k++) mem.add(g[i].members[k]);
    }
    String out; serializeJson(doc, out);
    _srv.send(200, "application/json", out);
}

// POST /api/groups — {op:"create", name, members:[addr...]} | {op:"leave", gid}
void AppWebServer::_handleGroupsPost()
{
    if (!_guard()) return;
    JsonDocument doc;
    if (deserializeJson(doc, _srv.arg("plain")) != DeserializationError::Ok) {
        _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"json\"}");
        return;
    }
    const char* op = doc["op"] | "";
    if (strcmp(op, "create") == 0) {
        const char* name = doc["name"] | "";
        // Адрес участника — полной ширины (mnmp_addr_t): при 2-байтной адресации
        // список группы обязан хранить >254, иначе показывает не тех.
        mnmp_addr_t members[GROUP_MEMBERS_MAX]; uint8_t cnt = 0;
        const long addrMax = (long)(mnmp_addr_t)~(mnmp_addr_t)0;
        for (JsonVariant v : doc["members"].as<JsonArray>()) {
            if (cnt >= GROUP_MEMBERS_MAX) break;
            long a = v.as<long>();
            if (a <= 0 || a >= addrMax) continue;         // 0=broadcast, max=gateway
            members[cnt++] = (mnmp_addr_t)a;
        }
        uint8_t gid = groupTable.create(name, members, cnt);
        if (!gid) { _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"full\"}"); return; }
        const Group* g = groupTable.find(gid);
        if (g) chatLog.sendGroupInfo(*g);   // разослать определение участникам
        char resp[40]; snprintf(resp, sizeof(resp), "{\"ok\":true,\"gid\":%u}", gid);
        _srv.send(200, "application/json", resp);
        return;
    }
    if (strcmp(op, "leave") == 0) {
        groupTable.leave((uint8_t)(doc["gid"] | 0));
        _srv.send(200, "application/json", "{\"ok\":true}");
        return;
    }
    _srv.send(400, "application/json", "{\"ok\":false,\"err\":\"op\"}");
}

void AppWebServer::_handleOtaUpload()
{
    HTTPUpload& up = _srv.upload();

    if (up.status == UPLOAD_FILE_START) {
        _otaHdrGot = 0; _otaHdrDone = false; _otaDenied = false; _otaResult = OTA_SHORT;
        if (!_authOk()) { _otaDenied = true; return; }
        // Ключ не прошит — отвергаем сразу, а не «принимаем и не проверяем».
        if (otaPubkeyUnset()) { _otaDenied = true; _otaResult = OTA_BAD_SIG; return; }
        return;
    }

    if (up.status == UPLOAD_FILE_ABORTED) { _otaApp.abort(); _otaDenied = true; return; }
    if (_otaDenied) return;                       // дочитываем тело в никуда

    if (up.status == UPLOAD_FILE_WRITE) {
        const uint8_t* p = up.buf;
        size_t n = up.currentSize;

        if (!_otaHdrDone) {                       // сперва набрать заголовок с подписью
            size_t need = sizeof(_otaHdrBuf) - _otaHdrGot;
            size_t take = n < need ? n : need;
            memcpy(_otaHdrBuf + _otaHdrGot, p, take);
            _otaHdrGot += (uint16_t)take;
            p += take; n -= take;
            if (_otaHdrGot < sizeof(_otaHdrBuf)) return;   // ждём остаток

            OtaHeader h;
            OtaResult r = otaParseHeader(_otaHdrBuf, _otaHdrGot, OTA_PUBKEY, &h);
            if (r == OTA_OK) r = otaCheckPolicy(h, OTA_BOARD, otaCurrentVersion(), otaMaxImageLen());
            if (r != OTA_OK) { _otaResult = r; _otaDenied = true; return; }
            if (!_otaApp.begin(h)) { _otaResult = OTA_TOO_BIG; _otaDenied = true; return; }
            _otaHdrDone = true;
        }
        if (n && !_otaApp.write(p, n)) { _otaResult = OTA_BAD_LEN; _otaDenied = true; }
        return;
    }

    if (up.status == UPLOAD_FILE_END) {
        _otaResult = _otaHdrDone ? _otaApp.finish() : OTA_SHORT;
    }
}

void AppWebServer::_handleOtaDone()
{
    if (!_guard()) return;                        // здесь уже можно ответить 401/403

    bool ok = (_otaResult == OTA_OK) && !_otaDenied;
    char body[192];
    snprintf(body, sizeof(body),
             "{\"ok\":%s,\"err\":\"%s\",\"got\":%u}",
             ok ? "true" : "false", ok ? "" : otaResultStr(_otaResult),
             (unsigned)_otaApp.written());
    _srv.send(ok ? 200 : 400, "application/json", body);

    if (ok) {
        // Ответ обязан уехать ДО перезагрузки, иначе браузер увидит обрыв и
        // покажет ошибку на успешном обновлении.
        _srv.client().flush();
        delay(300);
        ESP.restart();
    }
}

    // Позывное имя узла (профиль, персист в NVS). main читает для membership-анонса

    int8_t txPower() const { return _txPower; }

#if UAV_MAVLINK

    // правится POST /api/config {mavPeer}; 0 = не задан → мост только принимает.
    // main толкает значение в мост каждый loop (mavBridge.setPeer).
    mnmp_addr_t mavPeer() const { return _mavPeer; }

    // Последняя известная позиция БпЛА (GLOBAL_POSITION_INT #33, пришла через мост)
    // + курс/скорость для стрелки на карте (hdgDeg<0 = курса нет).
    // Отдаётся в /api/positions отдельным узлом карты (у node_table нет lat/lon).
    void setUavPos(mnmp_addr_t src, double lat, double lon, double altM,
                   float hdgDeg, float spdMps) {
        _uavAddr = src; _uavLat = lat; _uavLon = lon; _uavAlt = altM;
        _uavHdg = hdgDeg; _uavSpd = spdMps;
        _uavMs = (uint32_t)millis();
    }

    // Текущие радио-параметры для read-only показа в UI. main вызывает после

// ── Карта (Столп E, вариант A): офлайн-планшет в локальных метрах ENU ──
// Проекция как в nav/localize: восток = R·Δlon·cos(lat0), север = R·Δlat (R=6371000).
// Центр = свой фикс (иначе среднее якорей). Без тайлов/интернета.
//
// ТРЕКИ ЖИВУТ В БРАУЗЕРЕ, а не на устройстве: у ESP32 RAM в дефиците (26% и так занято),
// а история — чисто отображение. Точки копятся из того же опроса /api/positions, что
// идёт в общем цикле refresh() (поэтому трек растёт и когда открыта другая вкладка).
// ⚠ Плата за это: перезагрузка страницы обнуляет треки.
// КУРС: у БпЛА — из MAVLink #33 (поле hdg/скорости, авторитетный), у остальных узлов
// ВЫВОДИТСЯ из трека (пеленг между разнесёнными точками) — NODE_POS курса не несёт, и
// расширять его провод ради стрелки не стали.
var MAPDATA=null, TRACKS={}, MAPSEL=null, MAPXF=null, MAPPTS=[], MAPTRACKS=true, POSN=0;
var TRK_MAX=80, TRK_MIN_M=3, TRK_MIN_MS=4000, TRK_TTL_MS=900000;
var MAPPIN=false;                          // «приколоть» цель — тап мимо не сбрасывает выбор
var COMPASS_ON=false, DEVICE_HDG=null;     // задача 13: живой компас (прогресс-enhancement)

// Опрос позиций + накопление треков. Зовётся и из refresh() (всегда), и из mapLoad().
function posPoll(){
  return fetch('/api/positions').then(r=>r.json()).then(d=>{
    MAPDATA=d; var t=Date.now();
    if(d&&d.enabled){
      if(d.self&&d.self.fix)trackPush(d.self.addr,d.self.lat,d.self.lon,t);
      (d.nodes||[]).forEach(n=>trackPush(n.addr,n.lat,n.lon,t));
    }
    trackGc(t);
  }).catch(()=>{});
}
function trackPush(addr,lat,lon,t){
  if(addr==null||lat==null||lon==null)return;
  var a=TRACKS[addr]=TRACKS[addr]||[];
  if(a.length){
    var l=a[a.length-1];
    var d=enu(lat,lon,l.lat,l.lon), m=Math.hypot(d.e,d.n);
    if(m<TRK_MIN_M&&(t-l.t)<TRK_MIN_MS)return;         // стоит на месте — не копим шум
  }
  a.push({lat:lat,lon:lon,t:t});
  if(a.length>TRK_MAX)a.shift();
}
function trackGc(t){
  for(var k in TRACKS){var a=TRACKS[k];
    if(!a.length||(t-a[a.length-1].t)>TRK_TTL_MS)delete TRACKS[k];}
}
function trackClear(){TRACKS={};mapDraw();}
// Пеленг (град, север=0) по треку: ищем последнюю точку, разнесённую ≥5 м.
function trackHdg(addr){
  var a=TRACKS[addr]; if(!a||a.length<2)return null;
  var last=a[a.length-1];
  for(var i=a.length-2;i>=0;i--){
    var d=enu(last.lat,last.lon,a[i].lat,a[i].lon);
    if(Math.hypot(d.e,d.n)>=5){
      var deg=Math.atan2(d.e,d.n)*180/Math.PI;         // atan2(восток,север)
      return (deg+360)%360;
    }
  }
  return null;
}
function enu(lat,lon,lat0,lon0){const R=6371000,D=Math.PI/180;
  return {e:R*(lon-lon0)*D*Math.cos(lat0*D), n:R*(lat-lat0)*D};}

// Задача 11 (SIM-SYNTHESIS-REPORT.md §5.1): пеленг «от меня на цель» — НЕ курс
// движения цели (trackHdg выше — другое число), направление ОТ своей позиции
// К цели. Та же математика enu()+atan2.
function bearingTo(lat0,lon0,lat,lon){
  var d=enu(lat,lon,lat0,lon0);
  return (Math.atan2(d.e,d.n)*180/Math.PI+360)%360;
}
// Обязательное условие из находки: пеленг/дальность считаем ТОЛЬКО когда своя
// координата прошла анти-спуф (diag.gnssSpoof=0/OK или 1/NO_PREV — GNSS.spoof
// из /api/gnss, задача 4). Нет ГНСС-модуля вообще или прошивка без
// GNSS_ANTISPOOF (GNSS.spoof отсутствует) — гейту нечего проверять, старое
// поведение (доверяем, как раньше).
function selfTrusted(){
  if(!GNSS||!GNSS.enabled)return true;
  if(GNSS.spoof==null)return true;
  return GNSS.spoof===0||GNSS.spoof===1;
}
// Задача 12: «стоит на месте» — трек ЕСТЬ (≥2 точки), но trackHdg не нашёл
// смещения ≥5м ни на одной паре — отличаем от «данных ещё нет» (трек короче).
function isStationary(addr){
  var a=TRACKS[addr];
  return !!(a && a.length>=2 && trackHdg(addr)==null);
}

// открыть чат. Так случайный тап по карте не выкидывает оператора в переписку.
function mapTap(ev){
  if(!MAPXF||!MAPPTS.length)return;
  var cv=document.getElementById('mapcv'),r=cv.getBoundingClientRect();
  var mx=ev.clientX-r.left,my=ev.clientY-r.top,best=null,bd=1e9;
  MAPPTS.forEach(p=>{var d=Math.hypot(p.sx-mx,p.sy-my);if(d<bd){bd=d;best=p;}});
  if(!best||bd>22){if(!MAPPIN){MAPSEL=null;mapDraw();}return;}   // мимо всех — снять выбор (если не приколото)
  if(MAPSEL===best.addr&&!best.me){openConv('u:'+best.addr);return;}
  MAPSEL=best.addr;mapDraw();
}
// Панель выбранной точки: имя, дальность, курс/скорость, возраст, кнопка чата.
function mapInfo(p){
  var el=document.getElementById('map-info');if(!el)return;
  if(!p){el.style.display='none';el.innerHTML='';return;}
  el.style.display='';
  var dist=Math.hypot(p.x-((MAPPTS[0]&&MAPPTS[0].me)?MAPPTS[0].x:0),
                      p.y-((MAPPTS[0]&&MAPPTS[0].me)?MAPPTS[0].y:0));
  var bits=[];
  if(!p.me)bits.push('дальность '+fmtM(dist));
  // Задача 11: пеленг «от меня на цель» — ОТДЕЛЬНОЕ число от курса цели (ниже).
  if(!p.me&&p.brg!=null)bits.push('пеленг '+Math.round(p.brg)+'°');
  if(p.hdg!=null)bits.push('курс '+Math.round(p.hdg)+'°');
  else if(!p.me&&isStationary(p.addr))bits.push('стоит на месте');   // задача 12
  if(p.spd!=null&&p.spd>0)bits.push(p.spd.toFixed(1)+' м/с');
  if(p.alt!=null)bits.push('высота '+Math.round(p.alt)+' м');
  if(p.age!=null)bits.push('данные '+Math.round(p.age/1000)+' с назад');
  bits.push('коорд. '+p.lat.toFixed(6)+', '+p.lon.toFixed(6));
  var nm=p.me?'Я':((p.name&&p.name.trim())||hx(p.addr));
  // Задача 13: needle живого компаса (относительный пеленг цель-минус-устройство) —
  // элемент всегда в разметке, видимость/поворот выставляет updateCompassNeedle().
  var needle=(!p.me&&p.brg!=null)?'<span id="map-compass-needle" style="display:none;'

  el.innerHTML='<div style="display:flex;align-items:center;gap:10px">'
    +'<div class="grow"><div style="font-weight:600">'+(MAPPIN?'📌 ':'')+(p.uav?'✈ ':'')+esc(nm)+needle

    +'<button class="btn ghost" onclick="MAPSEL=null;MAPPIN=false;mapDraw()">✕</button></div>';

}

// Задача 13 (SIM-SYNTHESIS-REPORT.md §5.1): живой компас — ПРОГРЕСС-ENHANCEMENT
// поверх пеленга-текстом (задача 11), не замена: карта остаётся north-up, только
// маленькая стрелка в панели цели поворачивается на ОТНОСИТЕЛЬНЫЙ пеленг
// (курс на цель минус курс устройства) — «в какую сторону повернуться».
// ⚠ requestPermission() для DeviceOrientationEvent на iOS Safari работает
// ТОЛЬКО в HTTPS-контексте; точка доступа устройства — plain HTTP → на iPhone
// скорее всего не сработает вообще, на Android — версионно нестабильно.
// Поэтому ошибка/отказ разрешения НЕ ломает остальной UI — просто needle не
// появляется, текстовый пеленг остаётся основным способом (см. mapInfo).
function updateCompassNeedle(){
  var el=document.getElementById('map-compass-needle');
  if(!el)return;
  var sel=MAPPTS.find(function(p){return p.addr===MAPSEL;});
  if(!sel||sel.brg==null||DEVICE_HDG==null){el.style.display='none';return;}
  var rel=((sel.brg-DEVICE_HDG)%360+360)%360;
  el.style.display='inline-block';
  el.style.transform='rotate('+rel+'deg)';
}
function onDeviceOrientation(e){
  var h=(e.webkitCompassHeading!=null)?e.webkitCompassHeading:(e.alpha!=null?(360-e.alpha)%360:null);
  if(h==null)return;
  DEVICE_HDG=h;
  updateCompassNeedle();
}
function toggleCompass(){
  if(COMPASS_ON){
    COMPASS_ON=false;DEVICE_HDG=null;
    window.removeEventListener('deviceorientationabsolute',onDeviceOrientation);
    window.removeEventListener('deviceorientation',onDeviceOrientation);
    updateCompassNeedle();mapDraw();
    return;
  }
  if(typeof DeviceOrientationEvent==='undefined'){
    alert('Датчик ориентации недоступен на этом устройстве/браузере.');return;
  }
  var attach=function(){
    COMPASS_ON=true;
    if('ondeviceorientationabsolute' in window)window.addEventListener('deviceorientationabsolute',onDeviceOrientation);
    else window.addEventListener('deviceorientation',onDeviceOrientation);
    mapDraw();
  };
  if(typeof DeviceOrientationEvent.requestPermission==='function'){
    DeviceOrientationEvent.requestPermission().then(function(r){
      if(r==='granted')attach();
      else alert('Доступ к датчику ориентации не дан (нужен HTTPS — на точке доступа устройства это plain HTTP, скорее всего не сработает).');
    }).catch(function(){alert('Не удалось запросить доступ к датчику (обычно из-за отсутствия HTTPS).');});
  } else attach();
}
function niceStep(m){var p=Math.pow(10,Math.floor(Math.log10(m/3)));var n=m/3/p;
  return (n<1.5?1:n<3.5?2:n<7.5?5:10)*p;}
function fmtM(m){return m>=1000?(m/1000).toFixed(m<10000?1:0)+'км':Math.round(m)+'м';}

// ── Раскладка сообщений по разговорам ──
function rebuildMsgs(){
  MSG={};
  if(!ST||!ST.msgs)return;
  // msgs приходят новые->старые; для треда (reverse-column) оставляем как есть
  for(const m of ST.msgs){
    let key = m.gid ? ('g:'+m.gid) : (m.scope===1 ? ('u:'+m.addr) : 'general');
    (MSG[key]=MSG[key]||[]).push(m);
  }
}
function convPeers(){ // адреса личных разговоров (только ключи 'u:')
  return Object.keys(MSG).filter(k=>k[0]==='u').map(k=>+k.slice(2));
}

// ── Разговор по ключу: 'general' | 'u:<addr>' | 'g:<gid>' ──
function convFromKey(key){
  if(key==='general')return {...GENERAL};
  if(key[0]==='u'){const a=+key.slice(2);
    return {key,dst:a,gid:0,scope:1,addr:a,name:nameOf(a)};}
  const gid=+key.slice(2);const g=GROUPS.find(x=>x.gid===gid);
  return {key,dst:0,gid,scope:2,name:g?g.name:('Группа '+gid),members:g?g.members:[]};
}

// ── Список разговоров (вкладка Чаты): Общий + группы + личные ──
function renderConvList(){
  rebuildMsgs();
  let h=convItem('general');
  GROUPS.forEach(g=>{h+=convItem('g:'+g.gid)});
  const peers=new Set(convPeers());                      // только 'u:' ключи
  NODES.filter(n=>!n.self).forEach(n=>peers.add(n.addr));
  [...peers].sort((a,b)=>a-b).forEach(addr=>{h+=convItem('u:'+addr)});
  setHTML($('conv-list'), h||'<div class="empty">Нет чатов.</div>', 'convlist');
}
function convItem(key){
  const c=convFromKey(key); const msgs=MSG[key];
  const last=msgs&&msgs.length?msgs[0]:null;             // [0]=самое новое
  let prev=last?((last.out?'Вы: ':'')+esc(last.text)).slice(0,60):'Нет сообщений';

  if(key==='general'){col='#2e6f8f';ic='#';on=ST&&ST.synced;}
  else if(c.gid){col='#6a4fa0';ic='👥';on=true;
    if(!last)prev='группа · '+(c.members?c.members.length:0)+' уч.';}
  else{col=avColor(c.addr);ic=initial(c.name,c.addr);on=nodeOf(c.addr)&&online(nodeOf(c.addr));}

// «Видимость перегрузки» + «не дать усугубить затор»: ST.txq/txqMax (задача
// 14, /api/stats) — та же занятость MAC-очереди, что уже гейтит admission
// control; предыдущее исходящее сообщение ЭТОГО разговора ещё не доставлено
// (🕓, offline-outbox) — предупреждаем, но НЕ блокируем (может быть реально
// новое сообщение, а не случайный повтор).
function composerPending(){
  if(!CONV)return false;
  const list=MSG[CONV.key]||[];              // [0] = самое новое
  return !!(list.length && list[0].out && list[0].pend);
}
function updateComposerWarn(){
  const el=$('composer-warn');if(!el)return;
  const bits=[];
  if(ST&&ST.txqMax&&ST.txq>=Math.ceil(ST.txqMax/2))bits.push('⚠ сеть перегружена — сообщение может задержаться');
  if(composerPending())bits.push('🕓 предыдущее сообщение ещё не доставлено');
  el.textContent=bits.join(' · ');
  el.style.display=bits.length?'':'none';
}

  finally{

// UTC (григорианский) → Unix-секунды. Алгоритм Хиннанта (days_from_civil).
static int64_t daysFromCivil(int y, unsigned m, unsigned d)
{
    y -= (m <= 2);
    const int      era = (y >= 0 ? y : y - 399) / 400;
    const unsigned yoe = (unsigned)(y - era * 400);
    const unsigned doy = (153u * (m + (m > 2 ? -3 : 9)) + 2) / 5 + d - 1;
    const unsigned doe = yoe * 365 + yoe / 4 - yoe / 100 + doy;
    return (int64_t)era * 146097 + (int64_t)doe - 719468;
}

// Разбить NMEA-строку in-place по ',' и '*' (пустые поля сохраняются).
static int splitNmea(char* s, char** f, int maxf)
{
    int n = 0;
    f[n++] = s;
    for (char* p = s; *p && n < maxf; ++p) {
        if (*p == ',' || *p == '*') { *p = '\0'; f[n++] = p + 1; }
    }
    return n;
}

// "ddmm.mmmm" / "dddmm.mmmm" → градусы (degDigits = 2 для широты, 3 для долготы).
static float nmeaDegrees(const char* s, int degDigits)
{
    if (!s || !*s) return 0.0f;
    char deg[4] = {0};
    for (int i = 0; i < degDigits && s[i]; ++i) deg[i] = s[i];
    float d = (float)atoi(deg);
    float m = (float)atof(s + degDigits);
    return d + m / 60.0f;
}

void Gnss::begin()
{
    _instance = this;

    // Питание периферийной шины Heltec (Vext, GPIO36, active-LOW: LOW = вкл). На V4
    // GPS-модуль запитан через неё — без Vext L76K МОЛЧИТ (нет NMEA), даже если EN
    // подан. Раньше Vext не трогался вообще → вероятная причина «нет координат».
    // Радио (SX1262) на Heltec питается отдельно, Vext его не касается.
    pinMode(PIN_VEXT, OUTPUT); digitalWrite(PIN_VEXT, LOW);
    delay(50);

    // Питание/режим штатного модуля V4: EN активен LOW → подаём питание; снимаем
    // reset (HIGH = норма); STANDBY HIGH = форс-wake. Затем пауза на стабилизацию.
    pinMode(GNSS_EN_PIN, OUTPUT);      digitalWrite(GNSS_EN_PIN, LOW);
    pinMode(GNSS_RESET_PIN, OUTPUT);   digitalWrite(GNSS_RESET_PIN, HIGH);
    pinMode(GNSS_STANDBY_PIN, OUTPUT); digitalWrite(GNSS_STANDBY_PIN, HIGH);
    delay(GNSS_WARMUP_MS);

    // Автоскан RX-пина: распайка внешнего L76K на разных платах разная (память
    // отмечала неуверенность 38/39 vs 47/48). Ищем, на каком пине реально текут
    // NMEA, и слушаем именно его. Если нигде нет данных → проблема питания, не пина.
    _rxPin = _probeRxPin();

    gnssSerial.begin(GNSS_BAUD, SERIAL_8N1, _rxPin, GNSS_TX_PIN);
    // PPS L76K — push-pull выход; pulldown держит пин стабильным до появления
    // сигнала (нет ложных прерываний на старте / без фикса).
    pinMode(GNSS_PPS_PIN, INPUT_PULLDOWN);
    attachInterrupt(digitalPinToInterrupt(GNSS_PPS_PIN), _ppsIsr, RISING);
    Serial.printf("[GNSS] L76K UART%d rx=%d tx=%d pps=%d en=%d @%d\n",
                  GNSS_UART_NUM, _rxPin, GNSS_TX_PIN, GNSS_PPS_PIN, GNSS_EN_PIN, GNSS_BAUD);
}

// Перебор пинов-кандидатов под RX: на каждом ~1.3с (захватить 1-Гц всплеск NMEA)
// считаем байты и ищем '$'. '$' → точно NMEA, фиксируем пин. Иначе — пин с макс.
// числом байт (но это, скорее, не тот baud). Ноль везде → питание/модуль.
int Gnss::_probeRxPin()
{
    // 48/47 — стандарт Meshtastic для внешнего GPS на Heltec; далее прежние и
    // прочие свободные GPIO ESP32-S3 (минуя LoRa/FEM/LED/Vext/UART0/USB).
    static const int cand[] = { 48, 47, 38, 39, 33, 4, 5, 6, 1 };
    int bestPin = -1, bestCnt = 0;
    for (size_t i = 0; i < sizeof(cand) / sizeof(cand[0]); i++) {
        int rx = cand[i];
        gnssSerial.end(); delay(5);
        gnssSerial.begin(GNSS_BAUD, SERIAL_8N1, rx, GNSS_TX_PIN);
        uint32_t t0 = millis(); int cnt = 0; bool dollar = false;
        while (millis() - t0 < 1300) {
            while (gnssSerial.available()) { char c = (char)gnssSerial.read(); cnt++; if (c == '$') dollar = true; }
            if (dollar && cnt > 40) break;
        }
        Serial.printf("[GNSS] probe rx=%d: %d байт%s\n", rx, cnt, dollar ? " ($NMEA)" : "");
        if (dollar) { gnssSerial.end(); Serial.printf("[GNSS] RX=%d — есть NMEA, слушаем его\n", rx); return rx; }
        if (cnt > bestCnt) { bestCnt = cnt; bestPin = rx; }
    }
    gnssSerial.end();
    if (bestCnt >= 20) {
        Serial.printf("[GNSS] RX=%d: байты есть, но не NMEA — проверь baud модуля\n", bestPin);
        return bestPin;
    }
    Serial.println("[GNSS] НЕТ данных ни на одном пине → питание GPS (VCC/GND/EN) или модуль выключен");
    return GNSS_RX_PIN;
}

// Фронт 1PPS = граница UTC-секунды. Латчим точную локальную метку; целую секунду
// подставит ближайший последующий RMC (см. _parseLine).
void IRAM_ATTR Gnss::_ppsIsr()
{
    if (!_instance) return;
    _instance->_ppsLocalUs  = esp_timer_get_time();
    _instance->_ppsAwaiting = true;
}

void Gnss::poll()
{
    while (gnssSerial.available()) { _rawBytes++; _ingest((char)gnssSerial.read()); }

    // [ЗАКРЫТО: строк - 1; коммерческая тайна]
    if (_fixValid && (millis() - _lastFixMs) > GNSS_FIX_LOSS_MS) _fixValid = false;
}

void Gnss::_ingest(char c)
{
    if (c == '$') { _len = 0; _buf[_len++] = c; return; }
    if (c == '\n' || c == '\r') {
        if (_len > 6) { _buf[_len] = '\0'; _parseLine(_buf); }
        _len = 0;
        return;
    }
    if (_len < sizeof(_buf) - 1) _buf[_len++] = c;
}

bool Gnss::_checksumOk(const char* line) const
{
    if (line[0] != '$') return false;
    const char* star = strchr(line, '*');
    if (!star || !star[1] || !star[2]) return false;
    uint8_t sum = 0;
    for (const char* p = line + 1; p < star; ++p) sum ^= (uint8_t)*p;
    uint8_t want = (uint8_t)strtoul(star + 1, nullptr, 16);
    return sum == want;
}

void Gnss::_parseLine(char* line)
{
    if (!_checksumOk(line)) return;

    // Связь с модулем жива (нужно для теста в помещении: NMEA без фикса).
    _nmeaCount++;
    _lastNmeaMs = millis();

    char* f[20];
    int   n = splitNmea(line, f, 20);
    if (n < 1 || strlen(f[0]) < 6) return;
    const char* type = f[0] + 3;   // пропустить "$Gx"

    // ── RMC: время + дата + позиция + скорость, привязка PPS↔UTC ──────────────
    if (!strncmp(type, "RMC", 3) && n >= 10) {
        const char* tm     = f[1];     // hhmmss.sss
        const char  status = f[2][0];  // A=valid V=void
        if (status != 'A') { _fixValid = false; return; }
        if (strlen(tm) < 6 || strlen(f[9]) < 6) return;

        auto two = [](const char* p) { return (p[0]-'0')*10 + (p[1]-'0'); };
        int hh = two(tm), mm = two(tm + 2), ss = two(tm + 4);
        const char* dt = f[9];                       // ddmmyy
        int dd = two(dt), mo = two(dt + 2), yy = two(dt + 4);

        int64_t unixSec = daysFromCivil(2000 + yy, (unsigned)mo, (unsigned)dd) * 86400LL
                        + (int64_t)hh * 3600 + (int64_t)mm * 60 + ss;

        _lat      = nmeaDegrees(f[3], 2) * (f[4][0] == 'S' ? -1.0f : 1.0f);
        _lon      = nmeaDegrees(f[5], 3) * (f[6][0] == 'W' ? -1.0f : 1.0f);
        _speedMps = (float)atof(f[7]) * 0.514444f;   // узлы → м/с
        _fixValid  = true;
        _lastFixMs = millis();

        // Привязать целую UTC-секунду к латченному фронту PPS (он был ПЕРЕД этим
        // RMC). Чтение 64-бит метки — под защитой от ISR (неатомарно на 32-бит).
        if (_ppsAwaiting) {
            noInterrupts();
            int64_t ppsLocal = _ppsLocalUs;
            _ppsAwaiting     = false;
            interrupts();
            _pairPpsLocal  = ppsLocal;
            _pairUtcMicros = unixSec * 1000000LL;     // GPS_ASN_EPOCH_US=0 (от UTC 0)
            _pairReady     = true;
        }
        return;
    }

    // ── GGA: высота (для UAV-телеметрии; на время не влияет) ──────────────────
    if (!strncmp(type, "GGA", 3) && n >= 10) {
        _sats = (uint8_t)atoi(f[7]);                                // число спутников
        if (f[6][0] != '0' && f[9][0]) _altM = (float)atof(f[9]);   // fixQuality>0
        return;
    }
}

bool Gnss::takePps(int64_t& ppsLocalUs, int64_t& utcMicros)
{
    if (!_pairReady) return false;
    ppsLocalUs = _pairPpsLocal;
    utcMicros  = _pairUtcMicros;
    _pairReady = false;
    return true;
}

int64_t Gnss::utcNowMicros() const
{
    if (_pairUtcMicros == 0) return 0;
    return _pairUtcMicros + (timeCoreClock() - _pairPpsLocal);
}

//  суб-секундная дрожь NMEA не попадает в дисциплину.

    void begin();
    void poll();   // разбор UART; звать из loop()

    // Готовая пара PPS↔UTC (одноразовая выборка). true + заполняет out-параметры,
    // если со времени прошлого вызова появился новый сопоставленный фронт.
    bool takePps(int64_t& ppsLocalUs, int64_t& utcMicros);

    int64_t utcNowMicros() const;

    bool     fixValid()   const { return _fixValid; }
    float    lat()        const { return _lat; }
    float    lon()        const { return _lon; }
    float    altM()       const { return _altM; }
    float    speedMps()   const { return _speedMps; }
    uint32_t lastFixMs()  const { return _lastFixMs; }

private:

    void _ingest(char c);

    bool _checksumOk(const char* line) const;

    // состояние фикса/позиции
    bool     _fixValid  = false;
    float    _lat = 0, _lon = 0, _altM = 0, _speedMps = 0;
    uint32_t _lastFixMs = 0;

    // диагностика связи с модулем (NMEA идут даже без фикса)

    uint32_t _nmeaCount = 0;    // валидных (checksum-ok) NMEA-строк принято
    uint32_t _lastNmeaMs = 0;   // millis() последней такой строки

};

// CRC-16/MCRF4XX: полином 0x1021 в отражённой форме, init 0xFFFF, без финального XOR.
uint16_t mavlinkCrcAccumulate(const uint8_t* d, int n, uint16_t crc)
{
    if (!d) return crc;
    for (int i = 0; i < n; i++) {
        uint8_t tmp = (uint8_t)(d[i] ^ (uint8_t)(crc & 0xFF));
        tmp = (uint8_t)(tmp ^ (uint8_t)(tmp << 4));
        crc = (uint16_t)((crc >> 8) ^ ((uint16_t)tmp << 8) ^ ((uint16_t)tmp << 3)
                         ^ ((uint16_t)tmp >> 4));
    }
    return crc;
}

int mavlinkCrcExtra(uint32_t msgid)
{
    if (msgid > 255) return -1;                       // v2-диапазон вне таблицы
    uint8_t s = MAV_CRC_EXTRA[msgid];
    return (s == 0xFF) ? -1 : (int)s;
}

MavCrc mavlinkCheckFrame(const uint8_t* frame, int len)
{
    if (!frame || len < 8) return MavCrc::BAD;
    bool v2;
    if      (frame[0] == 0xFD) v2 = true;             // MAVLINK_STX_V2
    else if (frame[0] == 0xFE) v2 = false;            // MAVLINK_STX_V1
    else return MavCrc::BAD;

    int hdr    = v2 ? 9 : 5;                          // байт заголовка ПОСЛЕ STX
    int payLen = frame[1];
    if (len < 1 + hdr + payLen + 2) return MavCrc::BAD;   // кадр короче объявленного

    uint32_t msgid = v2 ? ((uint32_t)frame[7] | ((uint32_t)frame[8] << 8)
                                              | ((uint32_t)frame[9] << 16))
                        : (uint32_t)frame[5];
    int extra = mavlinkCrcExtra(msgid);
    if (extra < 0) return MavCrc::UNKNOWN_ID;

    // Сумма считается по кадру БЕЗ STX и без самих двух байт суммы, затем CRC_EXTRA.
    uint16_t crc = mavlinkCrcAccumulate(frame + 1, hdr + payLen, 0xFFFF);
    uint8_t  ex  = (uint8_t)extra;
    crc = mavlinkCrcAccumulate(&ex, 1, crc);

    const uint8_t* p = frame + 1 + hdr + payLen;      // сумма — little-endian
    uint16_t got = (uint16_t)((uint16_t)p[0] | ((uint16_t)p[1] << 8));
    return (crc == got) ? MavCrc::OK : MavCrc::BAD;
}

//  mavlink_crc.h  —  проверка контрольной суммы кадра MAVLink (Столп C, хвост 🟡).
//
//  ЗАЧЕМ. Парсер моста собирает кадр по полю длины и раньше принимал его НА ДОВЕРИИ.
//  UART до автопилота — единственное место, куда в наш путь входят битые данные (шлейф,
// [ЗАКРЫТО: строк - 2; коммерческая тайна]
//
//  КАК. CRC-16/MCRF4XX (init 0xFFFF) по кадру БЕЗ STX и без самой суммы, затем в ту же
//  сумму подмешивается байт CRC_EXTRA — хэш ОПРЕДЕЛЕНИЯ сообщения (имя+типы полей).
//  Он и ловит несовпадение диалектов: тот же msgid с другим набором полей → сумма не
// [ЗАКРЫТО: строк - 1; коммерческая тайна]
//  292 сообщения; в диапазон msgid 0..255 попали 188) — не выписана по памяти.
//
//  Подпись (v2 incompat_flags bit0) НЕ проверяем: 13 байт подписи идут ПОСЛЕ суммы,
//  на саму сумму не влияют; парсер их отбросит как шум (в них нет STX).
// =============================================================================

    OK = 0,          // сумма сошлась
    BAD = 1,         // не сошлась → кадр битый (или диалект расходится)
    UNKNOWN_ID = 2,  // msgid вне таблицы (msgid>255 или чужой диалект) → проверить нечем
};

uint16_t mavlinkCrcAccumulate(const uint8_t* d, int n, uint16_t crc);

ФАЙЛ 54. crc16.cpp - подсистема «защита информации»
НАЗНАЧЕНИЕ: вычисление контрольной суммы CRC-16.
==============================================================================
// =============================================================================
//  crc16.cpp  —  CRC-16/CCITT-FALSE (poly=0x1021, init=0xFFFF)
//  Adapted from ExpressLRS Crc2Byte.
// =============================================================================

#include "crc16.h"

CRC16 crc16;

CRC16::CRC16()
{
    const uint16_t poly    = 0x1021;
    const uint8_t  bitLen  = 16;
    const uint16_t highBit = 1 << (bitLen - 1);  // 0x8000

    for (uint16_t i = 0; i < 256; i++) {
        uint16_t crc = (uint16_t)(i << (bitLen - 8));
        for (uint8_t j = 0; j < 8; j++) {
            crc = (uint16_t)((crc << 1) ^ ((crc & highBit) ? poly : 0));
        }
        _table[i] = crc;
    }
}

uint16_t CRC16::calc(const uint8_t* data, size_t len, uint16_t crc) const
{
    while (len--) {
        crc = (uint16_t)((crc << 8) ^ _table[((crc >> 8) ^ (uint16_t)*data++) & 0x00FF]);
    }
    return crc;
}

ФАЙЛ 66. sha256.cpp - подсистема «защита информации»

// =============================================================================
//  sha256.cpp  —  см. sha256.h. FIPS 180-4 SHA-256 + HMAC + HKDF (RFC 5869).
//  Публичный алгоритм; портативная реализация (host + ESP32, один путь).
// =============================================================================
#include "sha256.h"
#include <string.h>

static inline uint32_t rotr(uint32_t x, uint32_t n) { return (x >> n) | (x << (32 - n)); }

static const uint32_t K[64] = {
    0x428a2f98,0x71374491,0xb5c0fbcf,0xe9b5dba5,0x3956c25b,0x59f111f1,0x923f82a4,0xab1c5ed5,
    0xd807aa98,0x12835b01,0x243185be,0x550c7dc3,0x72be5d74,0x80deb1fe,0x9bdc06a7,0xc19bf174,
    0xe49b69c1,0xefbe4786,0x0fc19dc6,0x240ca1cc,0x2de92c6f,0x4a7484aa,0x5cb0a9dc,0x76f988da,
    0x983e5152,0xa831c66d,0xb00327c8,0xbf597fc7,0xc6e00bf3,0xd5a79147,0x06ca6351,0x14292967,
    0x27b70a85,0x2e1b2138,0x4d2c6dfc,0x53380d13,0x650a7354,0x766a0abb,0x81c2c92e,0x92722c85,
    0xa2bfe8a1,0xa81a664b,0xc24b8b70,0xc76c51a3,0xd192e819,0xd6990624,0xf40e3585,0x106aa070,
    0x19a4c116,0x1e376c08,0x2748774c,0x34b0bcb5,0x391c0cb3,0x4ed8aa4a,0x5b9cca4f,0x682e6ff3,
    0x748f82ee,0x78a5636f,0x84c87814,0x8cc70208,0x90befffa,0xa4506ceb,0xbef9a3f7,0xc67178f2
};

static void sha256_block(uint32_t state[8], const uint8_t* p)
{
    uint32_t w[64];
    for (int i = 0; i < 16; i++)
        w[i] = ((uint32_t)p[i*4] << 24) | ((uint32_t)p[i*4+1] << 16) |
               ((uint32_t)p[i*4+2] << 8) | (uint32_t)p[i*4+3];
    for (int i = 16; i < 64; i++) {
        uint32_t s0 = rotr(w[i-15],7) ^ rotr(w[i-15],18) ^ (w[i-15] >> 3);
        uint32_t s1 = rotr(w[i-2],17) ^ rotr(w[i-2],19)  ^ (w[i-2] >> 10);
        w[i] = w[i-16] + s0 + w[i-7] + s1;
    }
    uint32_t a=state[0],b=state[1],c=state[2],d=state[3],
             e=state[4],f=state[5],g=state[6],h=state[7];
    for (int i = 0; i < 64; i++) {
        uint32_t S1 = rotr(e,6) ^ rotr(e,11) ^ rotr(e,25);
        uint32_t ch = (e & f) ^ (~e & g);
        uint32_t t1 = h + S1 + ch + K[i] + w[i];
        uint32_t S0 = rotr(a,2) ^ rotr(a,13) ^ rotr(a,22);
        uint32_t maj = (a & b) ^ (a & c) ^ (b & c);
        uint32_t t2 = S0 + maj;
        h=g; g=f; f=e; e=d+t1; d=c; c=b; b=a; a=t1+t2;
    }
    state[0]+=a; state[1]+=b; state[2]+=c; state[3]+=d;
    state[4]+=e; state[5]+=f; state[6]+=g; state[7]+=h;
}

void sha256_init(Sha256Ctx* c)
{
    c->state[0]=0x6a09e667; c->state[1]=0xbb67ae85; c->state[2]=0x3c6ef372; c->state[3]=0xa54ff53a;
    c->state[4]=0x510e527f; c->state[5]=0x9b05688c; c->state[6]=0x1f83d9ab; c->state[7]=0x5be0cd19;
    c->bitlen = 0; c->buflen = 0;
}

void sha256_update(Sha256Ctx* c, const uint8_t* data, size_t len)
{
    c->bitlen += (uint64_t)len * 8;
    while (len) {
        size_t take = SHA256_BLOCK_LEN - c->buflen;
        if (take > len) take = len;
        memcpy(c->buf + c->buflen, data, take);
        c->buflen += take; data += take; len -= take;
        if (c->buflen == SHA256_BLOCK_LEN) { sha256_block(c->state, c->buf); c->buflen = 0; }
    }
}

void sha256_final(Sha256Ctx* c, uint8_t out[SHA256_DIGEST_LEN])
{
    uint64_t bits = c->bitlen;
    uint8_t pad = 0x80;
    sha256_update(c, &pad, 1);
    uint8_t zero = 0;
    while (c->buflen != 56) sha256_update(c, &zero, 1);
    uint8_t lenbe[8];
    for (int i = 0; i < 8; i++) lenbe[i] = (uint8_t)(bits >> (56 - i*8));
    sha256_update(c, lenbe, 8);
    for (int i = 0; i < 8; i++) {
        out[i*4]   = (uint8_t)(c->state[i] >> 24);
        out[i*4+1] = (uint8_t)(c->state[i] >> 16);
        out[i*4+2] = (uint8_t)(c->state[i] >> 8);
        out[i*4+3] = (uint8_t)(c->state[i]);
    }
}

void sha256(const uint8_t* data, size_t len, uint8_t out[SHA256_DIGEST_LEN])
{
    Sha256Ctx c; sha256_init(&c); sha256_update(&c, data, len); sha256_final(&c, out);
}

// [ЗАКРЫТО: строк - 1; коммерческая тайна]
                 const uint8_t* data, size_t dataLen,
                 uint8_t out[SHA256_DIGEST_LEN])
{
    uint8_t k[SHA256_BLOCK_LEN] = {0};
    if (keyLen > SHA256_BLOCK_LEN) {
        sha256(key, keyLen, k);                  // длинный ключ → хешируем
    } else {
        memcpy(k, key, keyLen);
    }
    uint8_t ipad[SHA256_BLOCK_LEN], opad[SHA256_BLOCK_LEN];
    for (int i = 0; i < SHA256_BLOCK_LEN; i++) { ipad[i] = k[i] ^ 0x36; opad[i] = k[i] ^ 0x5c; }

    uint8_t inner[SHA256_DIGEST_LEN];
    Sha256Ctx c;
    sha256_init(&c); sha256_update(&c, ipad, SHA256_BLOCK_LEN);
    sha256_update(&c, data, dataLen); sha256_final(&c, inner);

    sha256_init(&c); sha256_update(&c, opad, SHA256_BLOCK_LEN);
    sha256_update(&c, inner, SHA256_DIGEST_LEN); sha256_final(&c, out);
}

void hkdf_sha256(const uint8_t* salt, size_t saltLen,
                 const uint8_t* ikm,  size_t ikmLen,
                 const uint8_t* info, size_t infoLen,
                 uint8_t* okm, size_t okmLen)
{
    // Extract: PRK = HMAC(salt, ikm). Пустая соль → 32 нуля (RFC 5869).
    uint8_t prk[SHA256_DIGEST_LEN];
    uint8_t zeroSalt[SHA256_DIGEST_LEN] = {0};
    if (!salt || saltLen == 0) { salt = zeroSalt; saltLen = SHA256_DIGEST_LEN; }
    // [ЗАКРЫТО: строк - 1; коммерческая тайна]

    // Expand: T(i) = HMAC(PRK, T(i-1)‖info‖i).
    uint8_t t[SHA256_DIGEST_LEN];
    size_t  tLen = 0;
    uint8_t counter = 1;
    size_t  done = 0;
    while (done < okmLen) {
        Sha256Ctx c;  // HMAC вручную для конкатенации T‖info‖counter
        uint8_t k[SHA256_BLOCK_LEN] = {0};
        memcpy(k, prk, SHA256_DIGEST_LEN);
        uint8_t ipad[SHA256_BLOCK_LEN], opad[SHA256_BLOCK_LEN];
        for (int i = 0; i < SHA256_BLOCK_LEN; i++) { ipad[i] = k[i] ^ 0x36; opad[i] = k[i] ^ 0x5c; }
        uint8_t inner[SHA256_DIGEST_LEN];
        sha256_init(&c); sha256_update(&c, ipad, SHA256_BLOCK_LEN);
        if (tLen) sha256_update(&c, t, tLen);
        if (infoLen) sha256_update(&c, info, infoLen);
        sha256_update(&c, &counter, 1);
        sha256_final(&c, inner);
        sha256_init(&c); sha256_update(&c, opad, SHA256_BLOCK_LEN);
        sha256_update(&c, inner, SHA256_DIGEST_LEN); sha256_final(&c, t);
        tLen = SHA256_DIGEST_LEN;

        size_t take = okmLen - done; if (take > SHA256_DIGEST_LEN) take = SHA256_DIGEST_LEN;
        memcpy(okm + done, t, take);
        done += take; counter++;
    }
}
